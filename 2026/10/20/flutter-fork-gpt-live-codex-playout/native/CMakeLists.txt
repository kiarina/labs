cmake_minimum_required(VERSION 3.16)
project(aec_engine LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(AEC_CORE "${CMAKE_CURRENT_LIST_DIR}/../vendor/webrtc-aec3/src" CACHE PATH "Prepared AEC3 extraction")
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "${AEC_CORE}/*.cc" "${AEC_CORE}/*.c" "${AEC_CORE}/*.cpp")
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64")
 list(FILTER SOURCES EXCLUDE REGEX ".*_(avx2|sse2|sse)\\.cc$")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64|ARM64")
 list(FILTER SOURCES EXCLUDE REGEX ".*_neon\\.cc$")
endif()
add_library(aec_engine SHARED aec_engine.cc playout_slot.cc playout_pipeline.cc ${SOURCES})
target_compile_definitions(aec_engine PRIVATE AEC_ENGINE_BUILD)
target_include_directories(aec_engine PUBLIC "${CMAKE_CURRENT_LIST_DIR}" PRIVATE
 "${AEC_CORE}" "${AEC_CORE}/api" "${AEC_CORE}/base/rtc_base" "${AEC_CORE}/base"
 "${AEC_CORE}/base/system_wrappers/include" "${AEC_CORE}/audio_processing" "${AEC_CORE}/absl"
 "${AEC_CORE}/base/jsoncpp/include" "${AEC_CORE}/audio_processing/aec3"
 "${AEC_CORE}/audio_processing/resampler" "${AEC_CORE}/audio_processing/logging"
 "${AEC_CORE}/audio_processing/include" "${AEC_CORE}/audio_processing/utility"
 "${AEC_CORE}/rtc_base/experiments" "${AEC_CORE}/common_audio"
 "${AEC_CORE}/common_audio/third_party/ooura/fft_size_128")
if(WIN32)
 target_compile_definitions(aec_engine PRIVATE WEBRTC_WIN NOMINMAX WIN32_LEAN_AND_MEAN)
 target_compile_options(aec_engine PRIVATE /utf-8)
 target_link_libraries(aec_engine PRIVATE winmm)
else()
 target_compile_definitions(aec_engine PRIVATE WEBRTC_POSIX)
 if(APPLE)
  target_compile_definitions(aec_engine PRIVATE WEBRTC_MAC)
  target_link_libraries(aec_engine PRIVATE "-framework CoreFoundation")
 else()
  target_compile_definitions(aec_engine PRIVATE WEBRTC_LINUX)
 endif()
endif()
if(ANDROID)
 target_compile_definitions(aec_engine PRIVATE WEBRTC_ANDROID)
 target_link_libraries(aec_engine PRIVATE log)
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64" AND NOT MSVC)
 foreach(SOURCE ${SOURCES})
  if(SOURCE MATCHES "_avx2\\.cc$")
   set_source_files_properties(${SOURCE} PROPERTIES COMPILE_OPTIONS "-mavx2")
  endif()
 endforeach()
endif()
option(AEC_BUILD_PROBE "Build the device-free slot/pipeline test" OFF)
if(AEC_BUILD_PROBE)
 add_executable(playout_probe playout_probe.cc)
 target_link_libraries(playout_probe PRIVATE aec_engine)
endif()
